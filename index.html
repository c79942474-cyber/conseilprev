# CONSEILPREV — Data.gouv.fr MCP Connector
import requests

class DataGouvFRConnector:
    def __init__(self):
        self.base_url = "https://www.data.gouv.fr/api/1"

    def search_datasets(self, query, page_size=20):
        r = requests.get(f"{self.base_url}/datasets/",
            params={"q": query, "page_size": page_size})
        return r.json()

connector = DataGouvFRConnector()
data = connector.search_datasets("intelligence artificielle")
print(f"Total: {data['total']} datasets")
# Flask — Route principale + API proxy
from flask import Flask, send_from_directory, jsonify, request
app = Flask(__name__, static_folder='.')

@app.route('/')
def index(): return send_from_directory('.', 'index.html')

@app.route('/api/datasets')
def datasets():
    q = request.args.get("query", "IA")
    return jsonify(connector.search_datasets(q))

# Filtres sectoriels
def filter_by_sector(datasets, sector):
    return [ds for ds in datasets
            if sector.lower() in
            (ds.get("title","") + ds.get("description","")).lower()]

sectors = ["santé", "énergie", "finance", "transport"]
for s in sectors:
    filtered = filter_by_sector(data["data"], s)
    print(f"[{s}] → {len(filtered)} datasets")
