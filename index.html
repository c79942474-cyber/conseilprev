# Installation : pip install requests flask flask-cors

import requests

class DataGouvFRConnector:
    def __init__(self, base_url="https://www.data.gouv.fr/api/1"):
        self.base_url = base_url
        self.session = requests.Session()

    def search_datasets(self, query, page=1, page_size=20):
        url = f"{self.base_url}/datasets/"
        params = {"q": query, "page": page, "page_size": page_size}
        response = self.session.get(url, params=params, timeout=10)
        return response.json()

    def get_dataset_resources(self, dataset_id):
        url = f"{self.base_url}/datasets/{dataset_id}/"
        response = self.session.get(url, timeout=10)
        return response.json().get("resources", [])

    def filter_by_sector(self, datasets, sector):
        return [ds for ds in datasets
                if sector.lower() in ds.get("title", "").lower()]

# Exemple d'utilisation
connector = DataGouvFRConnector()
datasets = connector.search_datasets("IA gouvernance")
ia_data = connector.filter_by_sector(datasets["data"], "intelligence")
print(f"Trouvé : {len(ia_data)} datasets IA")
# Flask API — Endpoints CONSEILPREV MCP

from flask import Flask, jsonify, request
from flask_cors import CORS

app = Flask(__name__)
CORS(app)
connector = DataGouvFRConnector()

@app.route("/api/datasets")
def get_datasets():
    query = request.args.get("query", "IA gouvernance")
    sector = request.args.get("sector", None)
    result = connector.search_datasets(query)
    if sector:
        result["data"] = connector.filter_by_sector(result["data"], sector)
    return jsonify(result)

@app.route("/api/datasets/<dataset_id>/resources")
def get_resources(dataset_id):
    return jsonify(connector.get_dataset_resources(dataset_id))

@app.route("/api/ia-compliance")
def get_ia_datasets():
    datasets = connector.get_ia_compliance_datasets()
    return jsonify({"data": datasets, "total": len(datasets)})

# Filtres sectoriels — CONSEILPREV use cases

sectors = ["Santé", "Énergie", "Finance", "Transport", "Éducation"]

def filter_datasets_by_sector(datasets, sector="Retail"):
    return [ds for ds in datasets
            if sector.lower() in
            (ds.get("title", "") +
             ds.get("description", "") +
             " ".join(ds.get("tags", []))).lower()]

def get_dataset_metrics(self, dataset_id):
    url = f"{self.base_url}/datasets/{dataset_id}/metrics/"
    response = self.session.get(url, timeout=10)
    return response.json()
