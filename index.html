# CONSEILPREV — Data.gouv.fr MCP Connector
import requests

class DataGouvFRConnector:
    def __init__(self):
        self.base = "https://www.data.gouv.fr/api/1"
        self.session = requests.Session()

    def search(self, query, page=1, page_size=15):
        r = self.session.get(f"{self.base}/datasets/", params={
            "q": query, "page": page,
            "page_size": page_size, "sort": "reuse_count"
        }, timeout=15)
        return r.json()

    def get_detail(self, dataset_id):
        r = self.session.get(f"{self.base}/datasets/{dataset_id}/")
        return r.json()

connector = DataGouvFRConnector()
data = connector.search("intelligence artificielle")
print(f"✅ {data['total']} datasets trouvés")

# Filtres sectoriels combinables
def filter_sector(datasets, sector, fmt=None):
    result = []
    for ds in datasets:
        text = (ds.get("title","") +
                ds.get("description","") +
                " ".join(ds.get("tags",[])))
        if sector.lower() not in text.lower(): continue
        if fmt:
            fmts = [r.get("format","").lower() for r in ds.get("resources",[])]
            if not any(fmt in f for f in fmts): continue
        result.append(ds)
    return result
